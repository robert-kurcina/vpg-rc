const fs=require('fs'),assert=require('node:assert/strict');
const T=require('../core/terrain_tiles');
const b=JSON.parse(fs.readFileSync('maps/leningrad.json','utf8'));
const svg=fs.readFileSync('exports-v302/leningrad.svg','utf8');
const audit=b.v302ImportAudit;
assert.equal(b.cells.length,197);
assert(!b.cells.some(c=>['2001','1801'].includes(c.id)),'Unsupported terrain cells persisted');
assert.equal(audit.sourceCoastSegments,53);
assert.equal(audit.registeredCoastSides,47);
assert.equal(audit.registeredLakeSides,9);
assert(audit.segmentSideAudit.every(e=>e.errorPx<1));
assert.equal(b.sourceRailRoutes.length,14);
assert.equal(b.preRegisteredRailGeometry,true);
assert.equal(b.authoritativeRailPolylineGeometry,true);
assert.equal((svg.match(/class="authored-center-rail"/g)||[]).length,14);
assert.equal((svg.match(/class="water-shore-fringe registered-shore-fringe"/g)||[]).length,56);
assert(!svg.includes('grid-face-rail-network'),'Obsolete inferred graph still renders rails');
assert(!svg.includes('data-hex="2001"'),'Unsupported cell rendered');
assert.equal(T.validate(b,{requireReciprocal:false}).errors.length,0);
console.log('PASS v302: authoritative Leningrad coast segments, exact railway polylines, source-only terrain footprint');
